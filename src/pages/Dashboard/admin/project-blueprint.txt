import { useState, useEffect } from "react";
import SchoolSetup from "./schoolSetup";
import CollegeSetup from "./collegeSetup";
import ImageUploader from "./ImageUploader";
import About from "./../about/about";

import {
  getMyInstitute,
  setAboutInstitute,
} from "../../../../api/instituteApi";

import styles from "./AboutInstitute.module.css";

function AboutInstitute() {
  const [type, setType] = useState("");
  const [name, setName] = useState("");
  const [about, setAbout] = useState("");

  const [logo, setLogo] = useState(null);
  const [building, setBuilding] = useState(null);
  const [gallery, setGallery] = useState([]);

  const [config, setConfig] = useState({});

  const [data, setData] = useState(null);
  const [editMode, setEditMode] = useState(false);

  const [loading, setLoading] = useState(true);

  /* ================= FETCH ================= */
  useEffect(() => {
    const fetchData = async () => {
      try {
        const d = await getMyInstitute();

        if (d) {
          setData(d);

          // 🔥 IMPORTANT: backend uses address
          setName(d.name || "");
          setAbout(d.address || "");
          setType(d.type || "");
          setLogo(d.logo || null);
          setBuilding(d.buildingImage || null);
          setGallery(d.gallery || []);
          setConfig(d.config || {});
        }
      } catch (err) {
        console.log("No institute found");
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, []);

  /* ================= VALIDATION ================= */
  const validate = () => {
    if (!name.trim()) return "Institute name required";
    if (!type) return "Select institute type";
    return null;
  };

  /* ================= SUBMIT ================= */
  const handleSubmit = async () => {
    const error = validate();
    if (error) {
      alert(error);
      return;
    }

    try {
      const res = await setAboutInstitute({
        name,
        about, // 🔥 maps to address in backend
        type,
        logo,
        buildingImage: building,
        gallery,
      });

      alert("Saved ✅");

      const updated = res;

      // SYNC STATE
      setData(updated);
      setEditMode(false);

      setName(updated.name || "");
      setAbout(updated.address || "");
      setType(updated.type || "");
      setLogo(updated.logo || null);
      setBuilding(updated.buildingImage || null);
      setGallery(updated.gallery || []);
      setConfig(updated.config || {});
    } catch (err) {
      console.error(err);
      alert(err.response?.data?.msg || "Error saving institute");
    }
  };

  /* ================= LOADING ================= */
  if (loading) {
    return <p style={{ padding: "20px" }}>Loading...</p>;
  }

  /* ================= VIEW MODE ================= */
  if (data && !editMode) {
    return (
      <div>
        <About data={data} />

        <div className={styles.editBtnWrapper}>
          <button
            className={styles.primaryBtn}
            onClick={() => setEditMode(true)}
          >
            Edit Institute
          </button>
        </div>
      </div>
    );
  }

  /* ================= FORM MODE ================= */
  return (
    <div className={styles.container}>
      <h2 className={styles.title}>
        {data ? "Edit Institute" : "Setup Your Institute"}
      </h2>

      <input
        placeholder="Institute Name"
        className={styles.name}
        value={name}
        onChange={(e) => setName(e.target.value)}
      />

      <textarea
        rows="6"
        placeholder="About your institute"
        className={styles.textarea}
        value={about}
        onChange={(e) => setAbout(e.target.value)}
      />

      {/* ================= IMAGES ================= */}
      <div className={styles.imageUpload}>
        <ImageUploader
          image={logo}
          setImage={setLogo}
          id="logoUpload"
          label="Institute Logo"
        />

        <ImageUploader
          image={building}
          setImage={setBuilding}
          id="buildingUpload"
          label="Building Image"
        />
      </div>

      {/* ================= TYPE ================= */}
      {!type && (
        <div className={styles.selector}>
          <button onClick={() => setType("school")} className={styles.btn}>
            School
          </button>

          <button onClick={() => setType("college")} className={styles.btn}>
            College / University
          </button>
        </div>
      )}

      {/* ================= SETUP ================= */}
      {type === "school" && (
        <SchoolSetup config={config} setConfig={setConfig} />
      )}

      {type === "college" && (
        <CollegeSetup config={config} setConfig={setConfig} />
      )}

      {/* ================= ACTION ================= */}
      <button className={styles.primaryBtn} onClick={handleSubmit}>
        {data ? "Update Institute" : "Save Institute"}
      </button>
    </div>
  );
}

export default AboutInstitute;

import { useEffect, useState } from "react";
import styles from "./collegeSetup.module.css";

/* ================= API ================= */
import { getMyInstitute } from "../../../../api/instituteApi";
import { getDepartments } from "../../../../api/departmentApi";
import { getCourses } from "../../../../api/courseApi";
import { getSubjects, createSubject } from "../../../../api/subjectApi";

import {
  createTeacher,
  createStudent,
} from "../../../../api/userAPI";

function CollegeSetup() {
  /* ================= STATE ================= */
  const [instituteId, setInstituteId] = useState("");

  const [departments, setDepartments] = useState([]);
  const [courses, setCourses] = useState([]);
  const [subjects, setSubjects] = useState([]);

  const [selectedDept, setSelectedDept] = useState("");

  const [loading, setLoading] = useState(false);

  const [subjectRows, setSubjectRows] = useState([]);
  const [teacherRows, setTeacherRows] = useState([]);
  const [studentRows, setStudentRows] = useState([]);

  /* ================= INIT ================= */
  useEffect(() => {
    init();
  }, []);

  const init = async () => {
    try {
      setLoading(true);

      const inst = await getMyInstitute();

      if (!inst || !inst._id) {
        throw new Error("Institute not found. Please login again.");
      }

      const instId = inst._id;
      setInstituteId(instId);

      // 🔥 Safe fetching (handles subject API mismatch)
      const [d, c] = await Promise.all([
        getDepartments(instId),
        getCourses({ instituteId: instId }),
      ]);

      let s = [];
      try {
        s = await getSubjects({ instituteId: instId });
      } catch {
        // fallback if backend expects departmentId instead
        s = [];
      }

      setDepartments(d || []);
      setCourses(c || []);
      setSubjects(s || []);

    } catch (err) {
      console.error("INIT ERROR:", err);

      if (err?.message) alert(err.message);
      else if (err?.response?.data?.msg) alert(err.response.data.msg);
      else alert("Failed to load initial data");

    } finally {
      setLoading(false);
    }
  };

  /* ================= SUBJECT ================= */
  const addSubjectRow = () => {
    setSubjectRows([...subjectRows, { name: "", code: "" }]);
  };

  const updateSubjectRow = (i, field, value) => {
    const copy = [...subjectRows];
    copy[i][field] = value;
    setSubjectRows(copy);
  };

  const saveSubjects = async () => {
    if (!selectedDept) return alert("Select Department");

    try {
      for (let row of subjectRows) {
        if (!row.name || !row.code) continue;

        await createSubject({
          name: row.name,
          code: row.code,
          departmentId: selectedDept,
        });
      }

      setSubjectRows([]);
      init();
    } catch (err) {
      console.error(err);
      alert("Failed to save subjects");
    }
  };

  /* ================= TEACHER ================= */
  const addTeacherRow = () => {
    setTeacherRows([
      ...teacherRows,
      { name: "", email: "", employeeId: "", subjects: [] },
    ]);
  };

  const updateTeacherRow = (i, field, value) => {
    const copy = [...teacherRows];
    copy[i][field] = value;
    setTeacherRows(copy);
  };

  const toggleSubject = (i, subjectId) => {
    const copy = [...teacherRows];
    const set = new Set(copy[i].subjects);

    set.has(subjectId) ? set.delete(subjectId) : set.add(subjectId);

    copy[i].subjects = [...set];
    setTeacherRows(copy);
  };

  const saveTeachers = async () => {
    if (!selectedDept) return alert("Select Department");

    try {
      for (let t of teacherRows) {
        if (!t.name || !t.email || !t.employeeId) continue;

        await createTeacher({
          ...t,
          departmentId: selectedDept,
        });
      }

      setTeacherRows([]);
      alert("Teachers saved");
    } catch (err) {
      console.error(err);
      alert("Failed to save teachers");
    }
  };

  /* ================= STUDENT ================= */
  const addStudentRow = () => {
    setStudentRows([
      ...studentRows,
      {
        name: "",
        email: "",
        rollNumber: "",
        semester: "",
        courseId: "",
      },
    ]);
  };

  const updateStudentRow = (i, field, value) => {
    const copy = [...studentRows];
    copy[i][field] = value;

  };

import { useState } from "react";
import styles from "./AboutInstitute.module.css";
import API from "./../../../../api/axios";

function ImageUploader({ image, setImage, id, label }) {
  const [loading, setLoading] = useState(false);

  const handleChange = async (e) => {
    const file = e.target.files[0];

    // ✅ Validate file
    if (!file || !file.type.startsWith("image/")) {
      alert("Please select a valid image file");
      return;
    }

    try {
      setLoading(true);

      // ✅ Instant preview
      const preview = URL.createObjectURL(file);
      setImage(preview);

      const formData = new FormData();

      // 🔥 CRITICAL FIX (must be "file")
      formData.append("file", file);

      // 🔥 CORRECT ENDPOINT (optional but better)
      const res = await API.post("/upload/image", formData);

      // ✅ Replace preview with cloud URL
      setImage(res.data.url);

    } catch (err) {
      console.error("UPLOAD ERROR:", err.response?.data || err.message);

      alert(err.response?.data?.msg || "Image upload failed");
      setImage(null);
    } finally {
      setLoading(false);
    }
  };

  const removeImage = () => {
    setImage(null);
  };

  return (
    <div className={styles.uploadBlock}>
      <p className={styles.uploadLabel}>{label}</p>

      <input
        type="file"
        accept="image/*"
        id={id}
        className={styles.hiddenInput}
        onChange={handleChange}
      />

      {!image ? (
        <label htmlFor={id} className={styles.uploadBox}>
          <span className={styles.uploadIcon}>⬆</span>
          <p>{loading ? "Uploading..." : "Upload"}</p>
        </label>
      ) : (
        <div className={styles.previewBox}>
          <img src={image} alt="preview" className={styles.previewImage} />

          <button
            type="button"
            className={styles.removeBtn}
            onClick={removeImage}
          >
            ✕
          </button>
        </div>
      )}
    </div>
  );
}

export default ImageUploader;

    let max = 0;

    else if (schoolType === "secondary") max = 10;
    else if (schoolType === "senior") max = 12;

        sections: prev[cls]?.sections || [],
        subjects: prev[cls]?.subjects || [],
        students: prev[cls]?.students || []
      };

      return {
        ...prev,
        [cls]: updater(base)
      };
    });

      ? list.filter(i => i !== val)
      : [...list, val];

function SubjectUI({ stored, teachers, onAdd }) {

  const [subId, setSubId] = useState("");
  const [teacherId, setTeacherId] = useState("");
  const [type, setType] = useState("compulsory");

  const available = GLOBAL_SUBJECTS.filter(
    s => !stored.some(x => x.code === s.code)
  );

  return (
    <div>
      <select onChange={e => setSubId(e.target.value)}>
        <option>Select Subject</option>
        {available.map(s => (
          <option key={s._id} value={s._id}>{s.name}</option>

        <option>Select Teacher</option>

        ))}
      </select>

      <select onChange={e => setType(e.target.value)}>
        {subjectTypes.map(t => (
          <option key={t}>{t}</option>
        ))}
      </select>

      <button onClick={() => {
        if (!subId || !teacherId) return;

        const sub = GLOBAL_SUBJECTS.find(s => s._id === subId);

          name: sub.name,

          type,
          teacherId: teacher.id,
          teacherName: teacher.name
        });

import { useEffect, useMemo, useState } from "react";
import { useAuth } from "../../../../context/authContext";
import StatsCard from "../../component/stats/stats/statsCard";
import styles from "./dashboard.module.css";

import { getMyInstitute } from "../../../../api/instituteApi";
import { getUsers } from "../../../../api/userAPI";
import { getCourses } from "../../../../api/courseApi";
import { getSubjects } from "../../../../api/subjectApi";
import { getClasses } from "../../../../api/classApi";
import { getRequests } from "../../../../api/requestApi";

function AdminDashboard() {
  const { user, token } = useAuth();
  const [data, setData] = useState({
    institute: null,
    students: [],
    teachers: [],
    courses: [],
    classes: [],
    subjects: [],
    requests: [],
  });
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (!token || !user?.instituteId) return;

    const load = async () => {
      try {
        const institute = await getMyInstitute();
        const [students, teachers, subjects, requests] = await Promise.all([
          getUsers({ role: "student" }),
          getUsers({ role: "teacher" }),
          getSubjects({ instituteId: user.instituteId, type: institute?.type }),
          getRequests(),
        ]);

        const courses =
          institute?.type === "college"
            ? await getCourses({ instituteId: user.instituteId })
            : [];
        const classes =
          institute?.type === "school"
            ? await getClasses(user.instituteId)
            : [];

        setData({ institute, students, teachers, courses, classes, subjects, requests });
      } catch (err) {
        console.error("Dashboard fetch error:", err);
      } finally {
        setLoading(false);
      }
    };

    load();
  }, [token, user?.instituteId]);

  const stats = useMemo(
    () => ({
      students: data.students.length,
      teachers: data.teachers.length,
      structure: data.institute?.type === "school" ? data.classes.length : data.courses.length,
      subjects: data.subjects.length,
      requests: data.requests.length,
    }),
    [data]
  );

  if (loading) return <div className={styles.loader}>Loading dashboard...</div>;

  return (
    <div className={styles.dashboard}>
      <div className={styles.header}>
        <div>
          <h1 className={styles.greeting}>Welcome, {user?.name?.split(" ")[0] || "Admin"}</h1>
          <p className={styles.subtitle}>{data.institute?.name || "Institute"} overview</p>
        </div>
      </div>

      <div className={styles.statsGrid}>
        <StatsCard title="Students" value={stats.students} color="blue" />
        <StatsCard title="Teachers" value={stats.teachers} color="green" />
        <StatsCard title={data.institute?.type === "school" ? "Classes" : "Courses"} value={stats.structure} color="purple" />
        <StatsCard title="Subjects" value={stats.subjects} color="orange" />
      </div>

      <div className={styles.mainGrid}>
        <section className={styles.card}>
          <h3>Pending Requests</h3>
          {data.requests.slice(0, 5).map((request) => (
            <div key={request._id} className={styles.item}>
              <span>{request.name}</span>
              <small>{request.role}</small>
            </div>
          ))}
          {data.requests.length === 0 && <p>No pending requests</p>}
        </section>

        <section className={styles.card}>
          <h3>Recent Students</h3>
          {data.students.slice(0, 5).map((student) => (
            <div key={student._id} className={styles.item}>
              <span>{student.name}</span>
              <small>{student.rollNumber || student.email}</small>
            </div>
          ))}
          {data.students.length === 0 && <p>No students found</p>}
        </section>

        <section className={styles.card}>
          <h3>Recent Teachers</h3>
          {data.teachers.slice(0, 5).map((teacher) => (
            <div key={teacher._id} className={styles.item}>
              <span>{teacher.name}</span>
              <small>{teacher.employeeId || teacher.email}</small>
            </div>
          ))}
          {data.teachers.length === 0 && <p>No teachers found</p>}
        </section>
      </div>
    </div>
  );
}
